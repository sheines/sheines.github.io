\begin{luacode}
function binom(n, p, k)
    local binom = 1
    k = math.floor(k + 0.5)
    if k > 0 then
        for i = 1, k do
            binom = binom * (n - i + 1) / i
        end
    end
    return binom * p^k * (1 - p)^(n - k)
end
\end{luacode}
